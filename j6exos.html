					//3.1 Importer les classes prédéfinie(Page & Locator)
					import { Page, Locator } from '@playwright/test';

					//3.2 Définir et Rendre la classe LoginPage disponible pour tous les fichiers
					export class LoginPage {
					  readonly page: Page; //Objet page: La page cible
					  readonly username: Locator; // elt cible( son selecteur css)
					  readonly password: Locator;
					  readonly submit: Locator;
					  readonly welcomeMessage: Locator;
					  //👉 readonly pour empêcher les tests de modifier les locators

					  constructor(page: Page) {
						this.page = page;
						this.username = page.locator('#username');
						this.password = page.locator('#password');
						this.submit = page.locator('#submit');
						this.welcomeMessage = page.locator('#welcome');
					  }
					  
					  // Se rendre à la page: pas de ".html" car c'est une route HTTP, pas un fichier 
					  async goto() {
						await this.page.goto('https://example.com/login');
					  }

					  // Remplir les champs et valider
					  async login(user: string, pass: string) {
						await this.username.fill(user);
						await this.password.fill(pass);
						await this.submit.click();
					  }
					}
			    

					import { test as base } from '@playwright/test';
					import { LoginPage } from '../pages/LoginPage';

					export const test = base.extend({
					  //const loginPage = new LoginPage(page => await use(loginPage);
					  loginPage: async ({ page }, use) => { await use(new LoginPage(page));}
					});

					export const expect = base.expect;
			    

					export function randomEmail() {
					  const id = Math.floor(Math.random() * 100000);
					  return `user${id}@test.com`;
					}
			    

					import { test, expect } from '../fixtures/fixtures';

					  const email = randomEmail();//génère une valeur aléatoire
					  await loginPage.login(email, 'password123');

					  await expect(loginPage.welcomeMessage).toBeVisible();
					});

🧱 3. Exo1: Menu

Objets: Tester les liens de navigation 


					Tu as des liens sous forme de boutons: Les “boutons” sont donc des <a>
					Tu veux vérifier que la classe ".activer" est sur le bon <a> selon la page,
					 et que les autres  ne l’ont pas.
					 
					Exemple: 
					6
					✔️ Le bouton actif a la classe : .activer
					✔️ Les autres boutons ont juste : .btn .popup, .btn .popup-bottom,  .default-btn
					✔️ Le numéro du bouton est le texte du lien:  <a ...>5</a>
					Donc on veut cibler les boutons par leur texte.

					Indications:
					✔️ Le numéro du bouton est le texte du lien:  <a ...>5</a>
						toHaveText(String(pageNumber))
					✔️ Le texte du lien commence par le numéro du bouton: <a ...>5abcd</a>
						toHaveText(new RegExp(`^${pageNumber}`))
					✔️ Le texte du lien commence par un espace ou un saut de ligne, suivi par  le numéro du bouton
					   toHaveText(new RegExp(`^\\s*${pageNumber}`))

					Question
					1.1 Ecrire le test de la première pagination: j3cours.html à J6cours.html
					1.2 Ecrire le test de la deuxième pagination: j3exos.html à J6exos.html

					Chaque test doit faire les 4 actions suivantes:
					🔷 1. Ouvrir chaque page:j3cours.html → j6cours.html
					🔷 2. Trouver le bouton actif .activer
					🔷 3. Vérifier que son texte correspond au numéro de la page
					🔷 4. Vérifie que les autres boutons n’ont pas .activer
			    

					// mkdir tests
					exemple1/
					│
					├── tests/
					│   └── navigation.spec.ts //→ Teste la navigation sur les cours
					|   └── navigation-exos.spec.ts //→ Teste la navigation sur les exos.
					
					// Routine de commandes
					npm init -y  // y(yest à tous): générer le fichier package.json sans poser aucune question.
					npm install -D @playwright/test // D(installer en devDependency): package développement, pas pour l’exécution en production.
					npx playwright install
			    

					import { test, expect } from '@playwright/test'; // usage objets de Page( exple page)

					test.describe('Pagination cours', () => {

					  for (let pageNumber = 3; pageNumber <= 6; pageNumber++) {
						test(`Sur j${pageNumber}cours.html, seul le lien ${pageNumber} est .activer`, async ({ page }) => {
							  // 🔷 1) Charger la page locale: Ouvrir chaque page(j3cours.html → j6cours.html)
							  await page.goto(`file:///D:/POEI/QA%20Automation/TypeScript2/j${pageNumber}cours.html`);
							  
							  // 🔷 2) Trouver le bouton actif .activer
							  // 2.1 Sélectionner tous les liens de pagination
							  const links = page.locator('.pagination a.btn'); // les 2 classes communes à tous les liens
							  // 2.2 Vérifier que le bon lien est actif
							  const activeLink = page.locator('.pagination a.activer');
							  
							  // 🔷 3) Vérifier que le lien actif contient le bon numéro de page(pageNumber)
							  //toHaveText(String(pageNumber));// String? converti en texte📌
							  //RegExp(`^3`)= commence par 3
							  //RegExp(`^\\s*3`)= commence par "un saut de ligne" suivi de 3
							  await expect(activeLink).toHaveText(new RegExp(`^\\s*${pageNumber}`)); //📌 accepte  espacement et retour à la ligne

							  // 🔷 4) Vérifier que les autres liens NE sont PAS actifs(dépourvus de la classe .activer)
							  for (let i = 3; i <= 6; i++) {
									//const link = page.locator(`.pagination a.btn`, { hasText: String(i) });// 1,2,3,4,5,6,7 Problème: 2 pagination <= collisions de sélecteurs
									const link = page.locator('nav.pagination:not(.pagination-blue) a.btn', { hasText: String(i) });// Exclure la 2ème pagination
									if (i === pageNumber) {
									  await expect(link).toHaveClass(/activer/); //📌 charche la classe .activer
									} else {
									  await expect(link).not.toHaveClass(/activer/); //📌 dit s'il n'a pas trouvé la classe .activer
									}
							  }
						});// test
					  }// for
					});// describe
					
					// Lancer le test
					npx playwright test
